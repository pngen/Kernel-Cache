# KernelCache examples.
function(kc_add_example name src)
  add_executable(${name} ${src})
  target_link_libraries(${name} PRIVATE KernelCache)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../distributed)
endfunction()
